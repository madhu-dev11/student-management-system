# HOW TO UPLOAD YOUR PROJECT TO GITHUB
# =====================================
# Follow these steps EXACTLY - one by one

# -----------------------------------------------
# STEP 1: Open Command Prompt / Terminal
# Go inside your project folder
# -----------------------------------------------

cd C:\path\to\student-management
# Example: cd C:\Users\Madhu\Desktop\student-management


# -----------------------------------------------
# STEP 2: Initialize Git in your project
# -----------------------------------------------

git init


# -----------------------------------------------
# STEP 3: Add all files
# -----------------------------------------------

git add .


# -----------------------------------------------
# STEP 4: Commit (save a snapshot)
# -----------------------------------------------

git commit -m "Initial commit - Student Management System"


# -----------------------------------------------
# STEP 5: Go to GitHub.com
# 1. Click the "+" button (top right)
# 2. Click "New repository"
# 3. Name it: student-management
# 4. Keep it Public
# 5. DO NOT check "Add README" (you already have one)
# 6. Click "Create repository"
# -----------------------------------------------


# -----------------------------------------------
# STEP 6: Copy the URL GitHub shows you
# It looks like: https://github.com/YourName/student-management.git
# Then run these commands:
# -----------------------------------------------

git remote add origin https://github.com/YOUR_USERNAME/student-management.git
git branch -M main
git push -u origin main


# -----------------------------------------------
# DONE! Refresh your GitHub page - files will be there!
# -----------------------------------------------


# -----------------------------------------------
# EVERY TIME YOU MAKE CHANGES, JUST RUN:
# -----------------------------------------------

git add .
git commit -m "describe what you changed"
git push
